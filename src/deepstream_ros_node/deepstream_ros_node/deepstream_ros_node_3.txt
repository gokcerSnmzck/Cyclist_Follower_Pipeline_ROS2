#!/usr/bin/env python3

import rclpy
from rclpy.node import Node
import subprocess
import threading


class DeepStreamAppRunner(Node):
    def __init__(self):
        super().__init__('deepstream_ros_node')
        self.get_logger().info("Starting DeepStream ROS node (running deepstream-app)...")

        path_to_yolo="/home/gokcer/DeepStream-Yolo/"

        config_path = "/home/gokcer/DeepStream-Yolo/deepstream_app_config.txt"

        self.process = subprocess.Popen(
            ["deepstream-app", "-c", config_path],
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            text=True,
            bufsize=1
        )

        # Start thread to stream logs
        self.thread = threading.Thread(target=self.read_logs, daemon=True)
        self.thread.start()

    def read_logs(self):
        for line in self.process.stdout:
            self.get_logger().info(line.strip())

    def destroy_node(self):
        self.get_logger().info("Shutting down deepstream-app...")
        self.process.terminate()
        super().destroy_node()


def main(args=None):
    rclpy.init(args=args)
    node = DeepStreamAppRunner()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
